{{- /* lazysizes and lightgallery, extended with Hugo Pipes responsive processing: - local raster images (page bundles and assets) get webp variants at 480/800/1200/1600 px and a width-based srcset (lazysizes picks the best fit via data-sizes="auto"); - the src falls back to the largest webp variant (≤ original width); - SVG and GIF are skipped (vector / animation would be lost); - explicit SrcSmall/SrcLarge callers keep the original density srcset. */ -}} {{- $src := .Src -}} {{- $width := .Width -}} {{- $height := .Height -}} {{- /* resolve the image resource: page bundles match by basename, global assets by path without leading slash (static/ is unresolvable) */ -}} {{- $u := urls.Parse $src -}} {{- $res := 0 -}} {{- if and $u.Path (not $u.Host) -}} {{- with .Resources.GetMatch (path.Base $u.Path) -}} {{- $res = . -}} {{- end -}} {{- if not $res -}} {{- with resources.Get (strings.TrimPrefix "/" $u.Path) -}} {{- $res = . -}} {{- end -}} {{- end -}} {{- with $res -}} {{- $src = .RelPermalink -}} {{- $width = $width | default .Width -}} {{- $height = $height | default .Height -}} {{- end -}} {{- end -}} {{- $small := .SrcSmall | default $src -}} {{- with dict "Path" .SrcSmall "Resources" .Resources | partial "function/resource.html" -}} {{- $small = .RelPermalink -}} {{- end -}} {{- $large := .SrcLarge | default $src -}} {{- with dict "Path" .SrcLarge "Resources" .Resources | partial "function/resource.html" -}} {{- $large = .RelPermalink -}} {{- end -}} {{- /* responsive webp variants via Hugo Pipes */ -}} {{- $srcset := "" -}} {{- if and $res $res.Width (not .SrcSmall) (not .SrcLarge) -}} {{- if in (slice "image/jpeg" "image/png" "image/webp" "image/tiff") $res.MediaType.Type -}} {{- $variants := slice -}} {{- range $w := slice 480 800 1200 1600 -}} {{- if lt $w $res.Width -}} {{- $v := $res.Resize (printf "%dx webp q75" $w) -}} {{- $variants = $variants | append (printf "%v %vw" $v.RelPermalink $w) -}} {{- end -}} {{- end -}} {{- $maxW := 1600 -}} {{- if lt $res.Width $maxW -}}{{- $maxW = $res.Width -}}{{- end -}} {{- $big := $res.Resize (printf "%dx webp q75" $maxW) -}} {{- $variants = $variants | append (printf "%v %vw" $big.RelPermalink $maxW) -}} {{- $src = $big.RelPermalink -}} {{- with $variants -}}{{- $srcset = delimit . ", " -}}{{- end -}} {{- end -}} {{- end -}} {{- $alt := .Alt | default .Title | default $src -}} {{- $loading := resources.Get "svg/loading.svg" | minify -}} {{- /* Eager: render the real image immediately (no lazysizes) and hint LCP */ -}} {{- $eager := .Eager -}} {{- $imgClass := "lazyload" -}} {{- $imgSrc := $loading.RelPermalink -}} {{- $imgSrcset := $srcset -}} {{- if $eager -}} {{- $imgClass = "" -}} {{- $imgSrc = $src -}} {{- if not $imgSrcset -}}{{- $imgSrcset = "" -}}{{- end -}} {{- else -}} {{- if not $imgSrcset -}}{{- $imgSrcset = printf "%v 1.5x, %v 2x" $small $large -}}{{- end -}} {{- end -}} {{- if .Linked -}} {{- else -}} {{- end -}}
